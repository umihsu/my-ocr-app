<script>

    const video = document.getElementById('video');

    const canvas = document.getElementById('canvas');

    const captureBtn = document.getElementById('btn-capture');

    const statusText = document.getElementById('status');

    const rawTextDisplay = document.getElementById('raw-text');

    const dateDisplay = document.getElementById('detected-date');


    // 1. 初始化相機（優先調用後鏡頭 environment）

    async function initCamera() {

        try {

            const stream = await navigator.mediaDevices.getUserMedia({

                video: { facingMode: 'environment' },

                audio: false

            });

            video.srcObject = stream;

        } catch (err) {

            statusText.innerText = "無法開啟相機，請檢查權限設定。";

            console.error("Camera error:", err);

        }

    }


    // 2. 點擊拍照並啟動 OCR 辨識

    captureBtn.addEventListener('click', async () => {

        // 將視訊畫面畫到 Canvas 上擷取圖片

        const context = canvas.getContext('2d');

        canvas.width = video.videoWidth;

        canvas.height = video.videoHeight;

        context.drawImage(video, 0, 0, canvas.width, canvas.height);

        

        // 取得圖片的 Base64 資料網址

        const dataUrl = canvas.toDataURL('image/jpeg');

        

        // 更新 UI 狀態

        captureBtn.disabled = true;

        statusText.innerText = "AI 正在辨識文字中，請稍候...";

        rawTextDisplay.innerText = "";

        dateDisplay.innerText = "掃描中...";


        try {

            // 3. 呼叫 Tesseract.js 進行英文與數字辨識 (eng)

            const result = await Tesseract.recognize(dataUrl, 'eng', {

                logger: m => {

                    if (m.status === 'recognizing text') {

                        statusText.innerText = `辨識進度: ${Math.round(m.progress * 100)}%`;

                    }

                }

            });


            const text = result.data.text;

            rawTextDisplay.innerText = text; // 顯示原始辨識文字

            statusText.innerText = "辨識完成！";


            // 4. 用正則表達式 (Regex) 撈出常見的日期格式 (如 2026.09.10 或 10/09/2026)

            // 這個 Regex 可以匹配 YYYY/MM/DD, DD/MM/YYYY, YYYY.MM.DD 等包含點、斜線或橫線的數字

            const dateRegex = /(\d{2,4}[.\/\-]\d{2}[.\/\-]\d{2,4})/g;

            const matches = text.match(dateRegex);


            if (matches && matches.length > 0) {

                // 如果有找到多個，先顯示第一個匹配到的日期

                dateDisplay.innerText = matches.join(', ');

            } else {

                dateDisplay.innerText = "未發現明顯日期格式，請調整角度再試一次。";

            }


        } catch (error) {

            statusText.innerText = "辨識過程中發生錯誤。";

            console.error("OCR Error:", error);

        } finally {

            captureBtn.disabled = false;

        }

    });


    // 網頁載入時自動開相機

    window.addEventListener('DOMContentLoaded', initCamera);

</script>
